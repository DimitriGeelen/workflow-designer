import sys, json
from playwright.sync_api import sync_playwright
B="http://192.168.10.107:3013"; out=sys.argv[1]
res={}
with sync_playwright() as p:
    br=p.chromium.launch()
    ctx=br.new_context(viewport={"width":1400,"height":900})
    pg=ctx.new_page()
    # Safety: every POST gets a garbage CSRF token, so nothing can mutate state
    def guard(route, req):
        if req.method=="POST":
            h=dict(req.headers); h["x-csrftoken"]="stale-reviewer-token"; h["x-csrf-token"]="stale-reviewer-token"
            route.continue_(headers=h)
        else: route.continue_()
    pg.route("**/*", guard)
    resps=[]
    pg.on("response", lambda r: resps.append((r.request.method,r.url,r.status,dict(r.headers))) if r.request.method=="POST" else None)
    pg.goto(B+"/approvals", wait_until="networkidle")
    ctx.clear_cookies()   # tab-1 session goes stale
    pg.evaluate("""() => htmx.ajax('POST','/approvals/reviewer-probe/approve',{source:document.body, swap:'none'})""")
    pg.wait_for_selector("#toast-container .wt-toast", timeout=5000)
    t=pg.locator("#toast-container .wt-toast").last
    res["toast_text"]=t.inner_text(); res["toast_class"]=t.get_attribute("class")
    res["toast_box"]=t.bounding_box()
    res["toast_scrollW_vs_clientW"]=t.evaluate("e=>[e.scrollWidth,e.clientWidth,getComputedStyle(e).color,getComputedStyle(e).backgroundColor,getComputedStyle(e).fontSize]")
    pg.screenshot(path=out+"/AC1-toast-full.png")
    t.screenshot(path=out+"/AC1-toast-closeup.png")
    res["htmx_post"]=[(m,u,s,h.get("hx-error-kind")) for m,u,s,h in resps]
    # Step 5: plain (non-htmx) POST navigation with stale token -> full page expected
    pg2=ctx.new_page(); pg2.route("**/*", guard)
    pg2.goto(B+"/approvals", wait_until="domcontentloaded"); ctx.clear_cookies()
    pg2.evaluate("""() => {const f=document.createElement('form');f.method='post';f.action='/approvals/reviewer-probe/approve';
      const i=document.createElement('input');i.name='_csrf_token';i.value='stale';f.appendChild(i);document.body.appendChild(f);
      HTMLFormElement.prototype.submit.call(f);}""")
    pg2.wait_for_load_state("networkidle")
    res["nav_title"]=pg2.title(); res["nav_url"]=pg2.url
    res["nav_html_len"]=len(pg2.content())
    res["nav_has_reload_btn"]=pg2.locator("button:has-text('Reload'), a:has-text('Reload')").count()
    res["nav_text_head"]=pg2.inner_text("body")[:400]
    pg2.screenshot(path=out+"/AC1-navigation-403-page.png", full_page=False)
    br.close()
print(json.dumps(res,indent=1,ensure_ascii=False))
